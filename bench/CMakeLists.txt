# The module's own native code and its vendored sources built for a Mac or a Linux machine, with a harness that
# runs it over recorded audio (the usage note at the top of harness.cc). Not part of the app builds; CI compiles
# it on Linux to prove the vendored tree and sources.cmake are complete on a clean machine.
cmake_minimum_required(VERSION 3.18)
project(rn_aec3_bench C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(MODULE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(WEBRTC_DIR "${MODULE_DIR}/vendor/webrtc-audio-processing/webrtc")
set(ABSL_DIR "${MODULE_DIR}/vendor/abseil-cpp")
include("${MODULE_DIR}/cpp/sources.cmake")

set(sources "${MODULE_DIR}/cpp/rn_aec3.cc")
foreach(source ${WEBRTC_SOURCES})
  list(APPEND sources "${WEBRTC_DIR}/${source}")
endforeach()
foreach(source ${ABSL_SOURCES})
  list(APPEND sources "${ABSL_DIR}/${source}")
endforeach()

set(defines WEBRTC_POSIX WEBRTC_LIBRARY_IMPL WEBRTC_APM_DEBUG_DUMP=0 NDEBUG)
if(APPLE)
  list(APPEND defines WEBRTC_MAC)
else()
  list(APPEND defines WEBRTC_LINUX)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  list(APPEND defines WEBRTC_ARCH_ARM64 WEBRTC_HAS_NEON)
  foreach(source ${WEBRTC_NEON_SOURCES})
    list(APPEND sources "${WEBRTC_DIR}/${source}")
  endforeach()
else()
  list(APPEND defines WEBRTC_ENABLE_AVX2)
  foreach(source ${WEBRTC_SSE2_SOURCES})
    list(APPEND sources "${WEBRTC_DIR}/${source}")
    set_source_files_properties("${WEBRTC_DIR}/${source}" PROPERTIES COMPILE_OPTIONS "-msse2")
  endforeach()
  foreach(source ${WEBRTC_AVX2_SOURCES})
    list(APPEND sources "${WEBRTC_DIR}/${source}")
    set_source_files_properties("${WEBRTC_DIR}/${source}" PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
  endforeach()
endif()

add_library(rn_aec3 STATIC ${sources})
target_include_directories(rn_aec3 PUBLIC "${WEBRTC_DIR}" "${ABSL_DIR}" "${MODULE_DIR}/cpp")
target_compile_definitions(rn_aec3 PUBLIC ${defines})
target_compile_options(rn_aec3 PRIVATE -O2 -w)

add_executable(harness harness.cc)
if(APPLE)
  target_link_libraries(harness rn_aec3 "-framework CoreFoundation")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(harness rn_aec3 Threads::Threads)
endif()
